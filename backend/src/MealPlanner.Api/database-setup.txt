DROP DATABASE IF EXISTS meal_planner_db;
CREATE DATABASE meal_planner_db;

-- ============================================================
-- Crear estructura y datos iniciales
-- Ejecutar conectado a meal_planner_db
-- ============================================================

CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'user',
    daily_calories INTEGER NOT NULL DEFAULT 2200 CHECK (daily_calories > 0),
    minimum_protein_grams INTEGER NOT NULL DEFAULT 120 CHECK (minimum_protein_grams >= 0),
    excluded_ingredients UUID[] NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE ingredients (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(150) NOT NULL UNIQUE,
    category VARCHAR(100) NOT NULL,
    calories_per_100g NUMERIC(8, 2) NOT NULL CHECK (calories_per_100g >= 0),
    protein_per_100g NUMERIC(8, 2) NOT NULL CHECK (protein_per_100g >= 0),
    carbs_per_100g NUMERIC(8, 2) NOT NULL DEFAULT 0 CHECK (carbs_per_100g >= 0),
    sugars_per_100g NUMERIC(8, 2) NOT NULL DEFAULT 0 CHECK (sugars_per_100g >= 0),
    fat_per_100g NUMERIC(8, 2) NOT NULL DEFAULT 0 CHECK (fat_per_100g >= 0),
    fiber_per_100g NUMERIC(8, 2) NOT NULL DEFAULT 0 CHECK (fiber_per_100g >= 0),
    is_gluten_free BOOLEAN NOT NULL DEFAULT FALSE,
    is_diabetic_friendly BOOLEAN NOT NULL DEFAULT FALSE,
    glycemic_index INTEGER CHECK (glycemic_index IS NULL OR glycemic_index BETWEEN 0 AND 100),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE recipes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL UNIQUE,
    meal_type VARCHAR(50) NOT NULL CHECK (meal_type IN ('breakfast', 'lunch', 'dinner', 'snack')),
    description TEXT,
    servings INTEGER NOT NULL CHECK (servings > 0),
    tags TEXT[] NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE recipe_ingredients (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    recipe_id UUID NOT NULL REFERENCES recipes(id) ON DELETE CASCADE,
    ingredient_id UUID NOT NULL REFERENCES ingredients(id),
    quantity NUMERIC(10, 2) NOT NULL CHECK (quantity > 0),
    unit VARCHAR(30) NOT NULL,
    UNIQUE (recipe_id, ingredient_id, unit)
);

CREATE TABLE user_saved_recipes (
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    recipe_id UUID NOT NULL REFERENCES recipes(id) ON DELETE CASCADE,
    saved_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (user_id, recipe_id)
);

CREATE TABLE inventory_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    ingredient_id UUID NOT NULL REFERENCES ingredients(id),
    quantity NUMERIC(10, 2) NOT NULL CHECK (quantity >= 0),
    unit VARCHAR(30) NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (user_id, ingredient_id, unit)
);

CREATE TABLE meal_plans (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    year INTEGER NOT NULL CHECK (year >= 2024),
    month INTEGER NOT NULL CHECK (month BETWEEN 1 AND 12),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE meal_plan_days (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    meal_plan_id UUID NOT NULL REFERENCES meal_plans(id) ON DELETE CASCADE,
    plan_date DATE NOT NULL,
    UNIQUE (meal_plan_id, plan_date)
);

CREATE TABLE planned_meals (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    meal_plan_day_id UUID NOT NULL REFERENCES meal_plan_days(id) ON DELETE CASCADE,
    recipe_id UUID NOT NULL REFERENCES recipes(id),
    meal_type VARCHAR(50) NOT NULL CHECK (meal_type IN ('breakfast', 'lunch', 'dinner', 'snack')),
    UNIQUE (meal_plan_day_id, meal_type)
);

CREATE INDEX idx_ingredients_name ON ingredients(name);
CREATE INDEX idx_recipes_meal_type ON recipes(meal_type);
CREATE INDEX idx_recipe_ingredients_recipe_id ON recipe_ingredients(recipe_id);
CREATE INDEX idx_recipe_ingredients_ingredient_id ON recipe_ingredients(ingredient_id);
CREATE INDEX idx_user_saved_recipes_user_id ON user_saved_recipes(user_id);
CREATE INDEX idx_inventory_items_user_id ON inventory_items(user_id);
CREATE INDEX idx_inventory_items_ingredient_id ON inventory_items(ingredient_id);
CREATE INDEX idx_meal_plans_user_id ON meal_plans(user_id);
CREATE INDEX idx_meal_plan_days_plan_id ON meal_plan_days(meal_plan_id);
CREATE INDEX idx_planned_meals_day_id ON planned_meals(meal_plan_day_id);

-- ============================================================
-- Consulta util: recetas con nutricion calculada
-- ============================================================

SELECT
    r.id,
    r.name,
    r.meal_type,
    r.servings,
    ROUND(SUM(i.calories_per_100g * ri.quantity / 100), 2) AS total_calories,
    ROUND(SUM(i.protein_per_100g * ri.quantity / 100), 2) AS total_protein,
    BOOL_AND(i.is_gluten_free) AS is_gluten_free,
    BOOL_AND(i.is_diabetic_friendly) AS is_diabetic_friendly
FROM recipes r
JOIN recipe_ingredients ri ON ri.recipe_id = r.id
JOIN ingredients i ON i.id = ri.ingredient_id
WHERE ri.unit = 'g'
GROUP BY r.id, r.name, r.meal_type, r.servings
ORDER BY r.name;
